371constexpr int32_t CountsToMilliC(int16_t counts) noexcept {
372 const int32_t scaled = static_cast<int32_t>(counts) * 1000; // at most 32 768 000
373 return (scaled >= 0) ? ((scaled + 64) / 128) : (-((-scaled + 64) / 128));

381constexpr Result<int16_t> MilliCToCounts(int32_t milli_c) noexcept {
382 if ((milli_c < kMinMilliC) || (milli_c > kMaxMilliC)) {
383 return Result<int16_t>::Failure(Error::InvalidArgument);
384 }
385 const int32_t scaled = milli_c * kCountsPerDegreeC; // at most 32 768 000 in magnitude
386 const int32_t counts = (scaled >= 0) ? ((scaled + 500) / 1000) : (-((-scaled + 500) / 1000));
387 return Result<int16_t>::Success(static_cast<int16_t>(counts)); // in range: checked above
388}

438constexpr Error Validate(const Config& config) noexcept {
439 if (!detail::IsEnumerator(config.mode) || !detail::IsEnumerator(config.averaging) ||
440 !detail::IsEnumerator(config.cycle_time) || !detail::IsEnumerator(config.alert_function) ||
441 !detail::IsEnumerator(config.alert_polarity) || !detail::IsEnumerator(config.alert_source)) {
442 return Error::InvalidArgument;
443 }
444 if (!MilliCToCounts(config.low_limit_mc).ok() || !MilliCToCounts(config.high_limit_mc).ok() ||
445 !MilliCToCounts(config.offset_mc).ok()) {
446 return Error::InvalidArgument;
447 }
448 if (config.low_limit_mc > config.high_limit_mc) {
449 return Error::InvalidArgument;
450 }
451 if ((config.reset_timeout_ms == 0U) || (config.reset_timeout_ms > kMaxTimeoutMs) ||
452 (config.eeprom_timeout_ms == 0U) || (config.eeprom_timeout_ms > kMaxTimeoutMs)) {
453 return Error::InvalidArgument;
454 }
455 return Error::Ok;
456}

458// The factory configuration is valid and equals the datasheet's reset values, proved at build time
459// (KB-CPP20-012): 6000h is 192 degC, 8000h is -256 degC (Table 7-1, p. 13).
460static_assert(Validate(Config{}) == Error::Ok);
461static_assert(MilliCToCounts(Config{}.high_limit_mc).value == 0x6000);
462static_assert(MilliCToCounts(Config{}.low_limit_mc).value == -32768);
463static_assert(CountsToMilliC(0x6000) == 192000);
464static_assert(CountsToMilliC(-32768) == -256000);
465static_assert(CountsToMilliC(0x7FFF) == kMaxMilliC);
466static_assert(!MilliCToCounts(kMaxMilliC + 1).ok() && !MilliCToCounts(kMinMilliC - 1).ok());
467static_assert(GetCycleTimeUs(CycleTime::S1, Averaging::X8) == 1000000U);
468static_assert(GetCycleTimeUs(CycleTime::Ms15p5, Averaging::X8) == 125000U);
469static_assert(GetOneShotWaitUs(Averaging::X1) == 35000U);
