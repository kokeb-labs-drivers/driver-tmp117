30// Optional hooks: a `Declares*` concept only tests the name, the `Has*` concept the exact
31// signature, `noexcept` and the result type; the driver rejects a hook that is declared but does
32// not match (KB-CPP20-007).

92[[nodiscard]] constexpr uint16_t ConfigWord(const Config& config) noexcept {
93 uint16_t word = reg::CfgMod::Insert(0_u16, ModBits(config.mode));
94 word = reg::CfgConv::Insert(word, static_cast<uint32_t>(ToUnderlying(config.cycle_time)));
95 word = reg::CfgAvg::Insert(word, static_cast<uint32_t>(ToUnderlying(config.averaging)));
96 uint32_t bits = word;
97 if (config.alert_function == AlertFunction::Threshold) {
98 bits |= reg::kCfgTherm;
99 }
100 if (config.alert_polarity == AlertPolarity::ActiveHigh) {
101 bits |= reg::kCfgPolarity;
102 }
103 if (config.alert_source == AlertSource::DataReady) {
104 bits |= reg::kCfgDrAlert;
105 }
106 return static_cast<uint16_t>(bits);
107}

111[[nodiscard]] constexpr uint16_t CountsWord(int32_t milli_c) noexcept {
112 return AsUnsigned16(MilliCToCounts(milli_c).value);

116[[nodiscard]] constexpr Status DecodeStatus(uint16_t word) noexcept {
117 Status s{};
118 const uint32_t bits = word;
119 s.high_alert = (bits & reg::kCfgHighAlert) != 0U;
120 s.low_alert = (bits & reg::kCfgLowAlert) != 0U;
121 s.data_ready = (bits & reg::kCfgDataReady) != 0U;
122 s.eeprom_busy = (bits & reg::kCfgEepromBusy) != 0U;
123 return s;
124}

126// The factory configuration encodes to the reset values of Table 7-3, p. 25.
127static_assert(ConfigWord(Config{}) == reg::kConfigurationReset);
128static_assert(CountsWord(Config{}.high_limit_mc) == reg::kHighLimitReset);
129static_assert(CountsWord(Config{}.low_limit_mc) == reg::kLowLimitReset);
130static_assert(CountsWord(Config{}.offset_mc) == reg::kTempOffsetReset);
131// Each field lands where the register map puts it.
132static_assert(ConfigWord(Config{.mode = Mode::Shutdown}) == 0x0620U);
133static_assert(ConfigWord(Config{.averaging = Averaging::X64, .cycle_time = CycleTime::S16}) ==
134 0x03E0U);
135static_assert(ConfigWord(Config{.alert_function = AlertFunction::Threshold,
136 .alert_polarity = AlertPolarity::ActiveHigh,
137 .alert_source = AlertSource::DataReady}) == 0x023CU);
138static_assert((ConfigWord(Config{}) & reg::kCfgMustBeZero) == 0U);
139static_assert(DecodeStatus(0xA000U).high_alert && DecodeStatus(0xA000U).data_ready &&
140 !DecodeStatus(0xA000U).low_alert && !DecodeStatus(0xA000U).eeprom_busy);
141static_assert(DecodeStatus(0x5000U).low_alert && DecodeStatus(0x5000U).eeprom_busy);
