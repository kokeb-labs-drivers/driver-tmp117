121 Result<uint16_t> ReadWord(uint8_t pointer) noexcept {
122 if (!reg::IsDefinedPointer(pointer)) {
123 return Result<uint16_t>::Failure(Error::InvalidArgument);
124 }
125 tx_[0] = pointer;
126 rx_[0] = 0U;
127 rx_[1] = 0U;
128 if (!port_.WriteRead(address_, tx_.data(), 1U, rx_.data(), 2U)) {
129 return Result<uint16_t>::Failure(Error::Bus);
130 }
131 const uint32_t word = (static_cast<uint32_t>(rx_[0]) << 8U) | static_cast<uint32_t>(rx_[1]);
132 return Result<uint16_t>::Success(static_cast<uint16_t>(word));
133 }
134
138 Result<void> WriteWord(uint8_t pointer, uint16_t value) noexcept {
139 if (!reg::IsWritablePointer(pointer)) {
140 return Result<void>::Failure(Error::InvalidArgument);
141 }
142 tx_[0] = pointer;
143 tx_[1] = static_cast<uint8_t>(static_cast<uint32_t>(value) >> 8U);
144 tx_[2] = static_cast<uint8_t>(static_cast<uint32_t>(value) & 0xFFU);
145 return port_.Write(address_, tx_.data(), kFrameSize) ? Result<void>::Success()
146 : Result<void>::Failure(Error::Bus);
147 }
148
150 Result<void> GeneralCallReset() noexcept {
151 tx_[0] = reg::kGeneralCallReset;
152 return port_.Write(reg::kGeneralCallAddress, tx_.data(), 1U)
153 ? Result<void>::Success()
154 : Result<void>::Failure(Error::Bus);
155 }
